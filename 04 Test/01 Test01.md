📦 Option 2: Manual Step-by-Step Crawling
Step 1: Setup Components
python
from datetime import datetime
from zoneinfo import ZoneInfo
from news_regime.config.models import DbConfig, CrawlConfig
from news_regime.storage.db import DbSessionManager
from news_regime.crawler.http_client import HttpClient
from news_regime.crawler.naver_client import NaverNewsClient
from news_regime.parser.mainnews_parser import MainNewsParser
from news_regime.parser.article_parser import ArticleParser
from news_regime.storage.repositories.news_repo import NewsRepository

# Setup
db_config = DbConfig(dsn="memory://")
session_manager = DbSessionManager(db_config)
session = session_manager.create_session()

crawl_config = CrawlConfig(base_url="https://finance.naver.com")
http_client = HttpClient(
    base_url=crawl_config.base_url,
    timeout=10.0,
    user_agent="news_regime_crawler"
)

# Create crawler
naver_client = NaverNewsClient(
    http_client=http_client,
    crawl_config=crawl_config,
    mainnews_parser=MainNewsParser(),
    article_parser=ArticleParser()
)

news_repo = NewsRepository()
Step 2: Crawl Main News List
python
# Crawl the main news page
now = datetime.now(ZoneInfo("Asia/Seoul"))
articles = naver_client.crawl_mainnews(crawled_at=now)

print(f"Found {len(articles)} articles")
for article in articles[:5]:  # Show first 5
    print(f"  - {article.title}")
    print(f"    URL: {article.url}")
    print(f"    Publisher: {article.publisher}")
Step 3: Fetch Full Article Details
python
# Crawl full article bodies
details = []
for meta in articles[:10]:  # First 10 articles
    try:
        # Skip if already crawled
        if news_repo.exists_by_url(session, meta.url):
            continue
        
        # Store metadata
        news_id = news_repo.insert_news_meta(session, meta)
        
        # Fetch full article
        detail = naver_client.crawl_article_detail(meta)
        news_repo.insert_news_detail(session, detail, news_id)
        details.append(detail)
        
        print(f"✓ Crawled: {meta.title}")
        print(f"  Body preview: {detail.body_text[:100]}...")
        
    except Exception as e:
        print(f"✗ Failed: {meta.url} - {e}")

print(f"\nCrawled {len(details)} articles successfully")

Method 2: Filter by Keyword

def search_news_by_keyword(session, keyword):
    """Search news by keyword in title or body."""
    results = []
    
    # Search in metadata (titles)
    for news_id, meta in session.get("news_meta", {}).items():
        if keyword.lower() in meta.title.lower():
            detail = session.get("news_detail", {}).get(news_id)
            results.append({
                'id': news_id,
                'title': meta.title,
                'url': meta.url,
                'publisher': meta.publisher,
                'published_at': meta.published_at,
                'body': detail.body_text if detail else None
            })
    
    return results

# Example: Search for Samsung news

📊 Complete Example: Crawl & Search
Create a file crawl_and_search.py:

python
#!/usr/bin/env python
"""Crawl Naver Finance news and search by keyword."""

from datetime import datetime, timedelta
from zoneinfo import ZoneInfo
from news_regime.cli.hourly_job import HourlyJob

def main():
    # Step 1: Crawl news
    print("=" * 60)
    print("STEP 1: Crawling Naver Finance News")
    print("=" * 60)
    
    job = HourlyJob()
    job.run_once()
    
    # Step 2: Access stored data
    print("\n" + "=" * 60)
    print("STEP 2: Searching Archived News")
    print("=" * 60)
    
    session = job.session_manager.create_session()
    
    # Search by keyword
    keyword = "삼성"  # Change this to your keyword
    print(f"\nSearching for: '{keyword}'")
    
    results = []
    for news_id, meta in session.get("news_meta", {}).items():
        if keyword in meta.title:
            results.append(meta)
    
    print(f"Found {len(results)} articles")
    
    for i, article in enumerate(results[:5], 1):
        print(f"\n{i}. {article.title}")
        print(f"   Publisher: {article.publisher}")
        print(f"   Published: {article.published_at}")
        print(f"   URL: {article.url}")
    
    # Show recommendations
    print("\n" + "=" * 60)
    print("STEP 3: Buy Recommendations")
    print("=" * 60)
    
    recommendations = session.get("recommendations", [])
    if recommendations:
        for i, rec in enumerate(recommendations[:5], 1):
            print(f"\n{i}. Symbol: {rec.symbol}")
            print(f"   Score: {rec.composite_score:.2f}")
            print(f"   Reason: {rec.reason_summary}")
    else:
        print("No recommendations generated (not enough positive news)")

if __name__ == "__main__":
    main()
